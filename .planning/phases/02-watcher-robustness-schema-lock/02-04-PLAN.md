<objective>
Implement the AUTH-05 refresh-token-death UX. When a refresh token dies (any of the four invalidation modes from PITFALLS.md Pitfall #1+#7, plus the OAuth-spec siblings unauthorized_client / invalid_client, plus repeated auth-flavored 403s after a refresh attempt), the watcher: (a) suspends all sheet writes, (b) turns the tray icon red, (c) surfaces a "Reauthorize" menu item that reopens the OAuth loopback flow against the same email, (d) on successful re-auth, replaces the wincred entry and resumes writes.

Purpose: AUTH-05 is the single biggest UX hole in Phase 1 — a Day-10 token-survival failure today produces a silent watcher stall. Phase 2 must surface the failure visibly within one upload cycle and give the guildie a one-click recovery.

Output: Typed IsRevokedRefreshToken helper; tray menu entry; suspend-on-death + resume-on-reauth state machine in app; OAuth re-runner that uses the existing email.
</objective>

    ```go
    package auth

    import (
        "errors"
        "strings"

        "golang.org/x/oauth2"
    )

    // IsRevokedRefreshToken returns true if err is the canonical Google
    // "this refresh token is permanently dead, prompt re-OAuth" shape. Covers
    // PITFALLS.md Pitfall #7's four invalidation modes (Testing-mode 7-day
    // expiry, six-month inactivity, user-revoked, password-change-with-revoke)
    // plus the OAuth-spec sibling errors that surface when the client itself
    // has been deleted/suspended.
    //
    // Detection priority:
    //   1. errors.As against *oauth2.RetrieveError + ErrorCode whitelist (typed path).
    //   2. Defensive string match (case-insensitive) for wrapping paths that
    //      surface a generic error whose message contains the OAuth code.
    func IsRevokedRefreshToken(err error) bool {
        if err == nil {
            return false
        }
        var re *oauth2.RetrieveError
        if errors.As(err, &re) {
            switch re.ErrorCode {
            case "invalid_grant", "unauthorized_client", "invalid_client":
                return true
            }
        }
        s := strings.ToLower(err.Error())
        if strings.Contains(s, "invalid_grant") ||
            strings.Contains(s, "unauthorized_client") ||
            strings.Contains(s, "invalid_client") {
            return true
        }
        return false
    }
    ```

    3. Create `internal/app/reauth.go`:

       ```go
       package app

       import (
           "context"
           "fmt"
           "log/slog"
           "net"
           "net/http"
           "sync/atomic"
           "time"

           "github.com/boejowen/SquireBot/internal/auth"
           "github.com/boejowen/SquireBot/internal/config"
           "github.com/boejowen/SquireBot/internal/tray"
       )

       const reauthorizeTimeout = 5 * time.Minute

       // Reauthorize re-runs the OAuth loopback flow against the existing email
       // stored in cfg.GoogleEmail. On success: writes the new refresh token to
       // wincred (replacing the dead one), clears authSuspended, returns the
       // refreshed TokenSource. The CALLER (tray click handler in main.go) is
       // responsible for restarting the watcher loop with the new TokenSource —
       // this function only handles the auth round-trip.
       //
       // Idempotent against rapid double-clicks: if a reauthorize attempt is
       // already in flight, the second call returns immediately with no error.
       func Reauthorize(ctx context.Context, cfg *config.Config, bc auth.BuildConstants, t *tray.Controller, authSuspended *atomic.Bool) error {
           if cfg.GoogleEmail == "" {
               return fmt.Errorf("Reauthorize: no GoogleEmail in config; wizard not yet run")
           }
           slog.Info("Reauthorize start", "email", cfg.GoogleEmail)
           t.SetStatus("Reauthorize: opening browser…")

           ln, err := net.Listen("tcp", "127.0.0.1:0")
           if err != nil {
               return fmt.Errorf("Reauthorize: listen: %w", err)
           }
           defer ln.Close()

           m := auth.NewManagerWithListener(cfg, bc, ln)
           mux := http.NewServeMux()
           m.AttachRoutes(mux)
           srv := &http.Server{Handler: mux, ReadHeaderTimeout: 10 * time.Second}
           go func() { _ = srv.Serve(ln) }()
           defer func() {
               shutCtx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
               defer cancel()
               _ = srv.Shutdown(shutCtx)
           }()

           if err := auth.OpenBrowser(m.AuthURL()); err != nil {
               slog.Warn("Reauthorize: open browser failed", "err", err)
           }

           timeoutCtx, cancel := context.WithTimeout(ctx, reauthorizeTimeout)
           defer cancel()

           res := m.RunOAuth(timeoutCtx)
           if res.Err != nil {
               slog.Error("Reauthorize: OAuth failed", "err", res.Err)
               t.SetStatus(fmt.Sprintf("Reauthorize failed: %v", res.Err))
               return res.Err
           }

           // Sanity: same email
           if res.Email != cfg.GoogleEmail {
               slog.Warn("Reauthorize: email changed",
                   "expected", cfg.GoogleEmail, "got", res.Email)
               // Persist new email and continue — better than dropping a successful auth.
               cfg.GoogleEmail = res.Email
               _ = cfg.Save()
           }

           // The Manager's exchangeAndStore already wrote the new refresh token
           // to wincred for cfg.GoogleEmail; no additional StoreToken call needed.

           authSuspended.Store(false)
           t.SetIconHealth(tray.HealthGreen)
           t.HideReauthorize()
           t.SetStatus(fmt.Sprintf("Reauthorized as %s — resumed", cfg.GoogleEmail))
           slog.Info("Reauthorize complete", "email", cfg.GoogleEmail)
           return nil
       }
       ```
