      export function holdersFor(itemId: number | null, viewRows: ViewRow[]): { char: string; count: number }[] {
        if (itemId === null) return [];                                  // custom want → "—" (D-07)
        const byChar = new Map<string, number>();
        for (const r of viewRows) {
          if (r.id !== itemId) continue;
          byChar.set(r.char, (byChar.get(r.char) ?? 0) + r.count);       // SUM per character
        }
        return [...byChar.entries()]
          .map(([char, count]) => ({ char, count }))
          .sort((a, b) => a.char.localeCompare(b.char));                 // SearchResults order
      }
