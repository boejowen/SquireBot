```typescript
export interface BankCoinForm {
  pp: number; gp: number; sp: number; cp: number;
}

export function showBankCoinSidebar(): void {
  const html = HtmlService.createHtmlOutput(`
    <div style="font-family:Arial,sans-serif;padding:12px;font-size:13px">
      <h3 style="margin-top:0">Bank Coin</h3>
      <p style="color:#666;font-size:11px">Manual entry — /outputfile inventory does not include coin totals.</p>
      <table style="width:100%">
        <tr><td>Platinum</td><td><input id="pp" type="number" min="0" style="width:100%"></td></tr>
        <tr><td>Gold</td><td><input id="gp" type="number" min="0" style="width:100%"></td></tr>
        <tr><td>Silver</td><td><input id="sp" type="number" min="0" style="width:100%"></td></tr>
        <tr><td>Copper</td><td><input id="cp" type="number" min="0" style="width:100%"></td></tr>
      </table>
      <button onclick="save()" style="margin-top:12px;padding:6px 14px">Save</button>
      <div id="msg" style="margin-top:10px;color:#080"></div>
      <script>
        google.script.run.withSuccessHandler(populate).getBankCoinForForm();
        function populate(v) {
          document.getElementById('pp').value = v.pp || 0;
          document.getElementById('gp').value = v.gp || 0;
          document.getElementById('sp').value = v.sp || 0;
          document.getElementById('cp').value = v.cp || 0;
        }
        function save() {
          const data = {
            pp: parseInt(document.getElementById('pp').value || '0', 10),
            gp: parseInt(document.getElementById('gp').value || '0', 10),
            sp: parseInt(document.getElementById('sp').value || '0', 10),
            cp: parseInt(document.getElementById('cp').value || '0', 10),
          };
          document.getElementById('msg').textContent = 'Saving...';
          google.script.run
            .withSuccessHandler(() => { document.getElementById('msg').textContent = 'Saved.'; })
            .withFailureHandler(e => { document.getElementById('msg').style.color = '#c00'; document.getElementById('msg').textContent = e.message; })
            .saveBankCoin(data);
        }
      </script>
    </div>
  `).setTitle('Set Bank Coin').setWidth(320);
  SpreadsheetApp.getUi().showSidebar(html);
}

export function getBankCoinForForm(): BankCoinForm {
  const meta = readMetaRows('_meta');
  const get = (key: string): number => {
    const row = meta.find(r => r.key === key);
    if (!row) return 0;
    const n = parseInt(row.value || '0', 10);
    return Number.isFinite(n) ? n : 0;
  };
  return {
    pp: get('bank_coin_pp'),
    gp: get('bank_coin_gp'),
    sp: get('bank_coin_sp'),
    cp: get('bank_coin_cp'),
  };
}

export function saveBankCoin(coin: BankCoinForm): void {
  for (const k of ['pp', 'gp', 'sp', 'cp'] as const) {
    const v = coin[k];
    if (!Number.isFinite(v) || v < 0) {
      throw new Error(`saveBankCoin: invalid ${k} value ${v}`);
    }
  }
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(30000)) throw new Error('saveBankCoin: lock_busy');
  try {
    writeMetaRow('_meta', 'bank_coin_pp', String(coin.pp));
    writeMetaRow('_meta', 'bank_coin_gp', String(coin.gp));
    writeMetaRow('_meta', 'bank_coin_sp', String(coin.sp));
    writeMetaRow('_meta', 'bank_coin_cp', String(coin.cp));
    writeMetaRow('_meta', 'bank_coin_last_updated', new Date().toISOString());
    log('info', 'saveBankCoin', { pp: coin.pp, gp: coin.gp, sp: coin.sp, cp: coin.cp });
  } finally {
    lock.releaseLock();
  }
  buildBank();  // outside the lock — buildBank acquires its own
}
```

### Task 3 — Range.protect on bank_coin cells

```typescript
export function protectBankCoinCells(): void {
  const sheet = getActiveSpreadsheet().getSheetByName('_meta');
  if (!sheet) return;
  const meta = readMetaRows('_meta');
  for (const key of ['bank_coin_pp', 'bank_coin_gp', 'bank_coin_sp', 'bank_coin_cp']) {
    const row = meta.find(r => r.key === key);
    if (!row) continue;
    const cell = sheet.getRange(row.rowIndex, 2);  // column B (value)
    // Check if already protected (idempotency)
    const existing = sheet.getProtections(SpreadsheetApp.ProtectionType.RANGE)
      .find(p => p.getRange().getA1Notation() === cell.getA1Notation()
               && p.getDescription() === 'SquireBot bank coin — edit via SquireBot menu');
    if (existing) continue;
    cell.protect()
      .setDescription('SquireBot bank coin — edit via SquireBot menu')
      .setWarningOnly(false);
  }
  log('info', 'protectBankCoinCells', { protected: 4 });
}
```

Call it:
- From migrateToV3 (post-success) — first-time setup.
- From installTriggers — defensive re-application (covers the case where a workbook was migrated before this code shipped).

### Task 4 — monitorCellCount trigger

`apps-script/src/triggers/monitorCellCount.ts`:

```typescript
import { log } from '../lib/log';
import { getActiveSpreadsheet, writeMetaRow } from '../lib/sheet-helpers';

const ALARM_THRESHOLD = 5_000_000;  // 50% of 10M cap
const HARD_CAP = 10_000_000;

export function monitorCellCount(): void {
  const ss = getActiveSpreadsheet();
  let total = 0;
  const perSheet: Array<{ name: string; cells: number }> = [];
  for (const sheet of ss.getSheets()) {
    const cells = sheet.getLastRow() * sheet.getLastColumn();
    perSheet.push({ name: sheet.getName(), cells });
    total += cells;
  }

  writeMetaRow('_status', 'cell_count', String(total));
  writeMetaRow('_status', 'cell_count_last_check', new Date().toISOString());

  if (total > ALARM_THRESHOLD) {
    const top5 = perSheet.sort((a, b) => b.cells - a.cells).slice(0, 5)
      .map(s => `${s.name}=${s.cells}`).join(', ');
    const err = {
      at: new Date().toISOString(),
      where: 'monitorCellCount',
      kind: 'cell_count_threshold',
      detail: `${total}/${HARD_CAP} (top: ${top5})`,
    };
    writeMetaRow('_meta', 'last_error', JSON.stringify(err));
    writeMetaRow('_status', 'last_error', JSON.stringify(err));
    log('warn', 'monitorCellCount', { total, threshold: ALARM_THRESHOLD, top5 });
  } else {
    log('info', 'monitorCellCount', { total, sheets: perSheet.length });
  }
}
```

**Tests** (`apps-script/src/__tests__/monitorCellCount.test.ts`):
- 3 small sheets total < threshold → cell_count written, no error.
- 3 sheets summing > 5M → cell_count written + last_error contains `cell_count_threshold` JSON.
- Top-5 reporting includes the 5 largest sheets in detail string.
- Empty workbook: cell_count = 0.

```typescript
export function installTriggers(): void {
  // existing delete-loop + first 4 trigger creates ...

  // NEW (3):
  ScriptApp.newTrigger('refreshWikiSpells').timeBased()
    .onWeekDay(ScriptApp.WeekDay.SUNDAY).atHour(4).inTimezone('America/Los_Angeles').create();
  // Note: refreshWikiSpells fires at 04:30, but Apps Script's atHour() doesn't have atMinute. Apps Script schedules within a 1-hour window for hourly+ triggers. Document expected fire-time as "04:00–05:00 PT Sunday".

  ScriptApp.newTrigger('monitorCellCount').timeBased()

    'SquireBot triggers installed (7 total).',
    '',
    '• onChange: rebuilds view + bank + spell_check + gear_check (debounced 10s)',
    '• 1h backstop: catches missed onChange events',
    '• Daily 03:00 PT: refreshPigparse',
    '• Sunday 03:00 PT: monitorCellCount',

    '• Sunday 05:00 PT: refreshWikiGearTier',

```

**Note on Apps Script trigger time precision:** atHour() schedules within a 1-hour window. Two triggers atHour(4) might both fire in the same window. We rely on each trigger's lock+debounce to handle concurrent runs (refreshWikiItems and refreshWikiSpells are independent — different sheets — so no lock contention; refreshWikiGearTier on a different hour anyway).
