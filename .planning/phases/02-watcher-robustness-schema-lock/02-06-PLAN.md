<objective>
Implement the OPS-04 auto-updater. On every cold start, the running binary first checks for a staged update (squirebot.exe.new + .expected-sha256 sidecar). If present and SHA-256 valid, it performs the minio/selfupdate startup-swap and re-execs into the new binary. Once running normally, a background goroutine fires every 24h: fetches latest.json from GitHub Releases, compares semver, downloads + SHA-256-verifies the new installer/binary if newer, stages it, and surfaces "update ready, restart to apply" in the tray. A new "Check for updates" tray menu item allows manual triggering.

Purpose: Without auto-update, every Phase 2/3/4/5 release requires guildies to manually re-run the SmartScreen walkthrough. With auto-update, a single tag push to GitHub propagates to every guildie's machine within 48h with no UX disruption beyond a tray status message + restart.

Output: New internal/update package; tray menu entry; cmd/squirebot/main.go startup hook; config.json gains a PendingUpdate boolean (informational; the file presence is the source of truth).
</objective>

       import (

       // Schema is locked at Phase 1; Phase 2's goreleaser migration (Plan 02-08)
       // preserves this exact set of fields.
       type Manifest struct {
           Version          string `json:"version"`            // "1.2.3" — no "v" prefix
           InstallerURL     string `json:"installer_url"`      // direct download URL for the installer .exe
           InstallerSHA256  string `json:"installer_sha256"`   // lowercase hex of the installer file
           ReleasedAt       string `json:"released_at"`        // RFC3339
       }

       // ManifestURL returns the canonical CDN URL for the latest manifest.
       // The "/releases/latest/download/<filename>" path is GitHub's stable
       // shortcut to the latest release's assets.
       func ManifestURL(owner, repo string) string {
           return fmt.Sprintf("https://github.com/%s/%s/releases/latest/download/latest.json", owner, repo)
       }

       // Fetch downloads + parses the latest.json. 5-second body timeout (well
       // below GitHub's typical CDN response time of <500ms).
       func Fetch(ctx context.Context, owner, repo string) (Manifest, error) {
           url := ManifestURL(owner, repo)
           req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
           if err != nil {
               return Manifest{}, fmt.Errorf("manifest req: %w", err)
           }
           req.Header.Set("User-Agent", "SquireBot-AutoUpdate")
           c := &http.Client{Timeout: 30 * time.Second}
           resp, err := c.Do(req)
           if err != nil {
               return Manifest{}, fmt.Errorf("manifest fetch %s: %w", url, err)
           }
           defer resp.Body.Close()
           if resp.StatusCode != http.StatusOK {
               return Manifest{}, fmt.Errorf("manifest fetch %s: HTTP %d", url, resp.StatusCode)
           }
           body, err := io.ReadAll(io.LimitReader(resp.Body, 4096))  // manifest is ~250 bytes
           if err != nil {
               return Manifest{}, fmt.Errorf("manifest read: %w", err)
           }
           var m Manifest
           if err := json.Unmarshal(body, &m); err != nil {
               return Manifest{}, fmt.Errorf("manifest parse: %w", err)
           }
           return m, nil
       }

       // IsNewer returns true if manifest > current using a 3-part numeric
       // semver comparison. Returns false on any parse failure (defensive —
       // a corrupt manifest must not trigger an update).
       func IsNewer(current, manifest string) bool {
           cParts, ok := parseVersion(current)
           if !ok {
               return false
           }
           mParts, ok := parseVersion(manifest)
           if !ok {
               return false
           }
           for i := 0; i < 3; i++ {
               if mParts[i] > cParts[i] {
                   return true
               }
               if mParts[i] < cParts[i] {
                   return false
               }
           }
           return false  // equal
       }

       func parseVersion(v string) ([3]int, bool) {
           v = strings.TrimPrefix(v, "v")
           parts := strings.Split(v, ".")
           if len(parts) != 3 {
               return [3]int{}, false
           }
           var out [3]int
           for i, p := range parts {
               n, err := strconv.Atoi(p)
               if err != nil {
                   return [3]int{}, false
               }
               out[i] = n
           }
           return out, true
       }
       ```

    import (
        "crypto/sha256"
        "encoding/hex"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "os"
        "path/filepath"
        "strings"

    // selfApplyFn is the package-level seam that wraps selfupdate.Apply.
    // Tests override it; production calls the real library.
    var selfApplyFn = selfupdate.Apply

    // Apply checks for a staged update (squirebot.exe.new + .expected-sha256
    // sidecar) adjacent to the running binary, verifies the SHA-256, and uses
    // minio/selfupdate to atomically swap the running .exe at startup.
    //
    // Returns (true, nil) on successful swap — caller MUST os.Exit(0) so the
    // new binary takes over via the next process launch.
    // Returns (false, nil) when no .new exists (the common path).
    // Returns (false, err) on any failure mode; caller logs and continues
    // running the OLD binary.
    func Apply() (swapped bool, err error) {
        exe, err := os.Executable()
        if err != nil {
            return false, fmt.Errorf("os.Executable: %w", err)
        }
        stagedPath := exe + ".new"
        hashPath := exe + ".expected-sha256"

        if _, err := os.Stat(stagedPath); errors.Is(err, os.ErrNotExist) {
            return false, nil
        }

        expected, err := os.ReadFile(hashPath)
        if err != nil {
            _ = os.Remove(stagedPath)
            return false, fmt.Errorf("read sidecar hash: %w", err)
        }
        expectedHex := strings.TrimSpace(string(expected))

        staged, err := os.Open(stagedPath)
        if err != nil {
            return false, fmt.Errorf("open staged: %w", err)
        }
        defer staged.Close()

        h := sha256.New()
        if _, err := io.Copy(h, staged); err != nil {
            return false, fmt.Errorf("hash staged: %w", err)
        }
        actualHex := hex.EncodeToString(h.Sum(nil))
        if actualHex != expectedHex {
            _ = os.Remove(stagedPath)
            _ = os.Remove(hashPath)
            return false, fmt.Errorf("staged hash mismatch: have %s, want %s", actualHex, expectedHex)
        }
        if _, err := staged.Seek(0, io.SeekStart); err != nil {
            return false, fmt.Errorf("rewind staged: %w", err)
        }

        if err := selfApplyFn(staged, selfupdate.Options{TargetPath: exe}); err != nil {
            return false, fmt.Errorf("selfupdate.Apply: %w", err)
        }

        _ = os.Remove(stagedPath)
        _ = os.Remove(hashPath)
        _ = os.Remove(filepath.Clean(exe + ".old"))

        slog.Info("auto-update applied", "exe", exe)
        return true, nil
    }
    ```

    package update

    import (
        "context"
        "crypto/sha256"
        "encoding/hex"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "os"
        "strings"
        "sync"
        "time"
    )

    // checkInterval is the cadence of automatic update checks. 24h matches
    // the heartbeat cadence; chosen for symmetry, not technical necessity.
    const checkInterval = 24 * time.Hour

    // sleepFn is overridable for testing.
    var sleepFn = realSleep
    func realSleep(ctx context.Context, d time.Duration) error {
        t := time.NewTimer(d)
        defer t.Stop()
        select {
        case <-ctx.Done(): return ctx.Err()
        case <-t.C: return nil
        }
    }

    // checkMu serializes CheckOnce calls — protects against the race where
    // the 24h tick fires while the user clicked "Check for updates".
    var checkMu sync.Mutex

    // StatusUpdater is a small callback the tray plumbs in; CheckOnce calls
    // it on a successful staging to surface the "Update ready" message.
    type StatusUpdater func(msg string)

    // CheckOnce performs a single update check + staging cycle.
    //   1. Fetch latest.json.
    //   2. If manifest.version <= current → return nil (no-op).
    //   3. If exePath+".new" already exists with matching sidecar SHA → return nil (already staged).
    //   4. Download manifest.installer_url to exePath+".new.tmp".
    //   5. SHA-256-verify against manifest.installer_sha256.
    //   6. On match: rename .tmp → .new; write .expected-sha256 sidecar; call statusFn.
    //   7. On mismatch: delete .tmp; return error.
    func CheckOnce(ctx context.Context, owner, repo, currentVersion, exePath string, statusFn StatusUpdater) error {
        checkMu.Lock()
        defer checkMu.Unlock()

        m, err := Fetch(ctx, owner, repo)
        if err != nil {
            return fmt.Errorf("CheckOnce manifest: %w", err)
        }
        if !IsNewer(currentVersion, m.Version) {
            slog.Info("auto-update: no newer version", "current", currentVersion, "manifest", m.Version)
            return nil
        }

        // Idempotency: if .new already exists with matching sidecar, skip download.
        stagedPath := exePath + ".new"
        hashPath := exePath + ".expected-sha256"
        if existingHash, err := os.ReadFile(hashPath); err == nil {
            if strings.TrimSpace(string(existingHash)) == m.InstallerSHA256 {
                slog.Info("auto-update: already staged", "version", m.Version)
                if statusFn != nil {
                    statusFn(fmt.Sprintf("Update ready (%s); restart to apply", m.Version))
                }
                return nil
            }
        }

        slog.Info("auto-update: downloading", "version", m.Version, "url", m.InstallerURL)

        // Download.
        req, err := http.NewRequestWithContext(ctx, http.MethodGet, m.InstallerURL, nil)
        if err != nil {
            return fmt.Errorf("CheckOnce download req: %w", err)
        }
        req.Header.Set("User-Agent", "SquireBot-AutoUpdate")
        c := &http.Client{Timeout: 5 * time.Minute}
        resp, err := c.Do(req)
        if err != nil {
            return fmt.Errorf("CheckOnce download: %w", err)
        }
        defer resp.Body.Close()
        if resp.StatusCode != http.StatusOK {
            return fmt.Errorf("CheckOnce download HTTP %d", resp.StatusCode)
        }

        tmpPath := exePath + ".new.tmp"
        f, err := os.Create(tmpPath)
        if err != nil {
            return fmt.Errorf("CheckOnce create tmp: %w", err)
        }
        h := sha256.New()
        mw := io.MultiWriter(f, h)
        if _, err := io.Copy(mw, resp.Body); err != nil {
            _ = f.Close()
            _ = os.Remove(tmpPath)
            return fmt.Errorf("CheckOnce copy: %w", err)
        }
        if err := f.Close(); err != nil {
            _ = os.Remove(tmpPath)
            return fmt.Errorf("CheckOnce close tmp: %w", err)
        }

        actualHex := hex.EncodeToString(h.Sum(nil))
        if actualHex != m.InstallerSHA256 {
            _ = os.Remove(tmpPath)
            return fmt.Errorf("CheckOnce hash mismatch: got %s, want %s", actualHex, m.InstallerSHA256)
        }

        // Atomic stage: rename .tmp → .new, then write sidecar.
        if err := os.Rename(tmpPath, stagedPath); err != nil {
            _ = os.Remove(tmpPath)
            return fmt.Errorf("CheckOnce stage: %w", err)
        }
        if err := os.WriteFile(hashPath, []byte(m.InstallerSHA256+"\n"), 0o600); err != nil {
            // Sidecar write failed — clean up .new so we don't half-stage.
            _ = os.Remove(stagedPath)
            return fmt.Errorf("CheckOnce sidecar: %w", err)
        }

        slog.Info("auto-update: staged", "version", m.Version, "sha256", actualHex)
        if statusFn != nil {
            statusFn(fmt.Sprintf("Update ready (%s); restart to apply", m.Version))
        }
        return nil
    }

    // RunDailyCheck blocks; fires CheckOnce immediately, then every checkInterval.
    // Errors are logged but never kill the goroutine — the watcher should keep
    // running even if GitHub is unreachable.
    func RunDailyCheck(ctx context.Context, owner, repo, currentVersion, exePath string, statusFn StatusUpdater) {
        if err := CheckOnce(ctx, owner, repo, currentVersion, exePath, statusFn); err != nil {
            slog.Warn("auto-update check failed", "err", err)
        }
        for {
            if err := sleepFn(ctx, checkInterval); err != nil {
                return
            }
            if err := CheckOnce(ctx, owner, repo, currentVersion, exePath, statusFn); err != nil {
                slog.Warn("auto-update check failed", "err", err)
            }
        }
    }
