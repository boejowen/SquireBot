export function didYouMean(query: string, itemNames: string[]): string[] {
  if (!query || !query.trim()) return [];          // 999.28 FIX — first line
  const q = query.toLowerCase();
  return itemNames.map((n)=>({n,d:levenshtein(q,n.toLowerCase())})).filter((x)=>x.d<=2&&x.d>0).sort((a,b)=>a.d-b.d).slice(0,3).map((x)=>x.n);
}
