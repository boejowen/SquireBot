**1a. `web/src/lib/items.ts` — add the pure `facetItems` sibling helper** (DOM-free — this is why it lives in `.ts` not `.svelte`; keeps the logic node-testable, Pitfall 5):
```ts
/** AND-combined Clicky/Haste facet (D-02): neither set → pass-through (full set);
 *  one set → that flag only; both set → the intersection. Pure; returns a new array. */
export function facetItems(
	rows: ItemRollup[],
	f: { clicky: boolean; haste: boolean }
): ItemRollup[] {
	return rows.filter((r) => (!f.clicky || r.is_clicky) && (!f.haste || r.has_haste));
}
```

export function searchCatalog(
