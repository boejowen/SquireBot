    ```go
    func (s *Store) ItemMasterIconStats(ctx context.Context) (map[int64]IconStats, error) {
        rows, err := s.db.QueryContext(ctx, `SELECT item_id, icon_id, statsblock FROM item_master`)
        if err != nil { return nil, err }
        defer rows.Close()
        out := make(map[int64]IconStats)
        for rows.Next() {
            var id int64
            var icon sql.NullInt64
            var stats sql.NullString
            if err := rows.Scan(&id, &icon, &stats); err != nil { return nil, err }
            out[id] = IconStats{IconID: icon.Int64, Statsblock: stats.String} // NULL → 0 / ""
        }
        return out, rows.Err()
    }
    ```

    ```go
    func buildItemRollups(viewRows []ViewRow, roster []store.RosterRow, iconStats map[int64]store.IconStats) []ItemRollup {
        flags := make(map[string]store.RosterRow, len(roster))
        for _, r := range roster { flags[r.Name] = r } // join holders → flags by char NAME
        byName := make(map[string]*ItemRollup)
        order := make([]string, 0)        // preserve first-seen order before the client re-sorts
        for _, vr := range viewRows {
            key := strings.ToLower(strings.TrimSpace(vr.Item)) // GROUP BY NORMALIZED NAME — never vr.ID
            roll := byName[key]
            if roll == nil {
                ic := iconStats[vr.ID]      // representative id-correct icon/stats (item_master EQ namespace)
                roll = &ItemRollup{
                    Name: vr.Item, Price: vr.Price, Prices: vr.Prices, WikiURL: vr.WikiURL,
                    WikiSummary: vr.WikiSummary, IsQuestItem: vr.IsQuestItem,
                    IconID: ic.IconID, Statsblock: ic.Statsblock,
                }
                byName[key] = roll
                order = append(order, key)
            }
            f := flags[vr.Char]
            isBank := f.IsBankToon || f.IsGuildBot
            roll.SummedQty += vr.Count
            if f.IsMine { roll.IsMine = true }
            roll.Holders = append(roll.Holders, ItemHolder{
                Char: vr.Char, SlotLabel: slotLabel(vr.Slot), Qty: vr.Count,
                LastSynced: vr.LastSynced, IsMine: f.IsMine, IsBank: isBank,
            })
        }
        // distinct holder count per rollup
        for key, roll := range byName {
            seen := make(map[string]struct{})
            for _, h := range roll.Holders { seen[h.Char] = struct{}{} }
            byName[key].HolderCount = int64(len(seen))
        }
        out := make([]ItemRollup, 0, len(order))
        for _, key := range order { out = append(out, *byName[key]) }
        return out
    }
    ```
