    ```go
    import (
        "context"
        "fmt"
        "log/slog"
        "time"

        "google.golang.org/api/sheets/v4"
    )

    // WriteHeartbeat performs a single batchUpdate that:
    //   1. Updates _char_owner.last_seen (column K) for every charName whose
    //      row exists in _char_owner.
    //   2. Upserts a _status row keyed on (owner_email, char_name) with
    //      watcher_version (col C), last_heartbeat (col F).
    //
    // The whole operation is ONE BatchUpdate call (Pitfall D / quota economy
    // â€” one batchUpdate, one quota debit). Cells are StringValue. fields="userEnteredValue".
    //
    // charNames not present in _char_owner are skipped for the _char_owner
    // update but still get a _status row (a heartbeat may fire before a new
    // character has had its first inventory upload).
    func (c *Client) WriteHeartbeat(ctx context.Context, ownerEmail string, charNames []string, watcherVersion string) error {
        if c.spreadsheetID == "" {
            return fmt.Errorf("WriteHeartbeat: spreadsheetID not set")
        }
        if len(charNames) == 0 {
            return nil
        }
        now := time.Now().UTC().Format(time.RFC3339)

        // 1. Look up _char_owner row indices keyed by char_name.
        coSheetID, err := c.EnsureSheet(ctx, "_char_owner")
        if err != nil {
            return fmt.Errorf("ensure _char_owner: %w", err)
        }
        coVR, err := c.valuesGet(ctx, "_char_owner!A:A")
        if err != nil {
            return fmt.Errorf("read _char_owner: %w", err)
        }
        // Map of char_name â†’ 0-indexed row in _char_owner (index 0 is header).
        coRows := map[string]int{}
        for i, row := range coVR.Values {
            if i == 0 || len(row) < 1 {
                continue
            }
            name, _ := row[0].(string)
            if name != "" {
                coRows[name] = i  // 0-indexed; col K = column index 10
            }
        }

        // 2. Look up _status rows keyed by (owner_email, char_name).
        statusSheetID, err := c.EnsureSheet(ctx, "_status")
        if err != nil {
            return fmt.Errorf("ensure _status: %w", err)
        }
        statusVR, err := c.valuesGet(ctx, "_status!A:B")
        if err != nil {
            return fmt.Errorf("read _status: %w", err)
        }
        statusRows := map[string]int{}  // key = "email|char" â†’ row index
        for i, row := range statusVR.Values {
            if i == 0 || len(row) < 2 {
                continue
            }
            email, _ := row[0].(string)
            name, _ := row[1].(string)
            statusRows[email+"|"+name] = i
        }

        // 3. Build a single BatchUpdate that does ALL the writes.
        var requests []*sheets.Request

        // 3a. _char_owner.last_seen updates (col K, index 10).
        for _, charName := range charNames {
            rowIdx, ok := coRows[charName]
            if !ok {
                slog.Info("heartbeat: char missing from _char_owner; skipping last_seen update", "char", charName)
                continue
            }
            requests = append(requests, &sheets.Request{
                UpdateCells: &sheets.UpdateCellsRequest{
                    Range: &sheets.GridRange{
                        SheetId:          coSheetID,
                        StartRowIndex:    int64(rowIdx),
                        EndRowIndex:      int64(rowIdx + 1),
                        StartColumnIndex: 10,  // K = last_seen
                        EndColumnIndex:   11,
                    },
                    Rows:   []*sheets.RowData{toRowData([]string{now})},
                    Fields: "userEnteredValue",
                },
            })
        }

        // 3b. _status row updates / appends.
        // Existing rows: PARTIAL in-place UpdateCells covering ONLY cols A, B, C, F.
        // Cols D (last_inventory_upload) and E (last_spellbook_upload) are populated
        // by WriteInventory / WriteSpellbook -- the heartbeat MUST NOT touch them
        // or it would clobber the freshness signal on every 24h tick (CONTEXT.md
        // schema-lock: extend-only, never overwrite a non-heartbeat-owned cell).
        // Implementation: emit THREE narrow UpdateCellsRequest entries per existing row
        // (A:A, B:C, F:F). For missing rows, AppendCells with a 6-cell row where
        // D and E are empty strings -- this is correct because D/E haven't been
        // written yet, so we're not clobbering anything. WriteInventory /
        // WriteSpellbook will populate D/E on their first run for that char.
        for _, charName := range charNames {
            key := ownerEmail + "|" + charName
            if rowIdx, ok := statusRows[key]; ok {
                // Partial update: A (owner_email), B+C (char_name + watcher_version), F (last_heartbeat).
                // Three separate UpdateCellsRequest blocks so D and E are physically untouched.
                aSv := ownerEmail
                requests = append(requests, &sheets.Request{
                    UpdateCells: &sheets.UpdateCellsRequest{
                        Range: &sheets.GridRange{
                            SheetId:          statusSheetID,
                            StartRowIndex:    int64(rowIdx),
                            EndRowIndex:      int64(rowIdx + 1),
                            StartColumnIndex: 0, // A
                            EndColumnIndex:   1,
                        },
                        Rows: []*sheets.RowData{{Values: []*sheets.CellData{
                            {UserEnteredValue: &sheets.ExtendedValue{StringValue: &aSv}},
                        }}},
                        Fields: "userEnteredValue",
                    },
                })
                bSv, cSv := charName, watcherVersion
                requests = append(requests, &sheets.Request{
                    UpdateCells: &sheets.UpdateCellsRequest{
                        Range: &sheets.GridRange{
                            SheetId:          statusSheetID,
                            StartRowIndex:    int64(rowIdx),
                            EndRowIndex:      int64(rowIdx + 1),
                            StartColumnIndex: 1, // B
                            EndColumnIndex:   3, // through C (exclusive end)
                        },
                        Rows: []*sheets.RowData{{Values: []*sheets.CellData{
                            {UserEnteredValue: &sheets.ExtendedValue{StringValue: &bSv}},
                            {UserEnteredValue: &sheets.ExtendedValue{StringValue: &cSv}},
                        }}},
                        Fields: "userEnteredValue",
                    },
                })
                fSv := now
                requests = append(requests, &sheets.Request{
                    UpdateCells: &sheets.UpdateCellsRequest{
                        Range: &sheets.GridRange{
                            SheetId:          statusSheetID,
                            StartRowIndex:    int64(rowIdx),
                            EndRowIndex:      int64(rowIdx + 1),
                            StartColumnIndex: 5, // F
                            EndColumnIndex:   6,
                        },
                        Rows: []*sheets.RowData{{Values: []*sheets.CellData{
                            {UserEnteredValue: &sheets.ExtendedValue{StringValue: &fSv}},
                        }}},
                        Fields: "userEnteredValue",
                    },
                })
            } else {
                // Append a new row. D and E are empty strings here -- this is fine
                // because the row didn't exist before, so there's nothing to clobber.
                row := []string{ownerEmail, charName, watcherVersion, "", "", now}
                cells := make([]*sheets.CellData, 0, 6)
                for _, s := range row {
                    sv := s
                    cells = append(cells, &sheets.CellData{
                        UserEnteredValue: &sheets.ExtendedValue{StringValue: &sv},
                    })
                }
                requests = append(requests, &sheets.Request{
                    AppendCells: &sheets.AppendCellsRequest{
                        SheetId: statusSheetID,
                        Rows:    []*sheets.RowData{{Values: cells}},
                        Fields:  "userEnteredValue",
                    },
                })
            }
        }
        if len(requests) == 0 {
            return nil
        }

        _, err = c.batchUpdate(ctx, &sheets.BatchUpdateSpreadsheetRequest{Requests: requests})
        if err != nil {
            return fmt.Errorf("WriteHeartbeat batchUpdate: %w", err)
        }
        return nil
    }
    ```

    ```go
    package heartbeat

    import (
        "context"
        "log/slog"
        "sync/atomic"
        "time"

        "github.com/boejowen/SquireBot/internal/config"
        "github.com/boejowen/SquireBot/internal/sheet"
    )

    // Interval is the cadence between heartbeat fires. 24h interval (NOT
    // wall-clock daily) per CONTEXT.md decision. time.Timer self-reschedule
    // (NOT time.Ticker, NOT robfig/cron) per RESEARCH.md ## Don't Hand-Roll.
    const Interval = 24 * time.Hour

    // sleepFn is the package-level sleep used for testing. Production uses
    // realSleep (which honors ctx cancellation via timer + select). Tests
    // override to a controllable shim.
    var sleepFn = realSleep
    func realSleep(ctx context.Context, d time.Duration) error {
        t := time.NewTimer(d)
        defer t.Stop()
        select {
        case <-ctx.Done():
            return ctx.Err()
        case <-t.C:
            return nil
        }
    }

    // Run blocks until ctx is cancelled. Fires WriteHeartbeat once on entry,
    // then every Interval. On suspended state (authSuspended==true), skips
    // the API call but keeps scheduling the next tick (so heartbeat resumes
    // automatically after re-auth without a watcher restart).
    //
    // The active-char roster is derived from cfg.LastKnownInventoryMtime keys
    // (which is the watcher's authoritative "characters this guildie owns"
    // signal: a char appears there as soon as it has had at least one
    // successful inventory upload). Spellbook-only characters are picked up
    // via cfg.LastKnownSpellbookMtime as well â€” union both maps.
    func Run(ctx context.Context, sc *sheet.Client, cfg *config.Config, ownerEmail, watcherVersion string, authSuspended *atomic.Bool) {
        tick := func() {
            if authSuspended != nil && authSuspended.Load() {
                slog.Info("heartbeat skipped: auth suspended")
                return
            }
            charNames := activeChars(cfg)
            if err := sc.WriteHeartbeat(ctx, ownerEmail, charNames, watcherVersion); err != nil {
                slog.Warn("heartbeat write failed", "err", err, "chars", len(charNames))
                // Non-fatal: WATCH-07 retry already happened inside WriteHeartbeat.
                // If auth died, the next tick will see authSuspended set by the
                // makeOnInventoryChange path (or by retrying on its own).
            } else {
                slog.Info("heartbeat written", "chars", len(charNames))
            }
        }

        // Immediate first fire.
        tick()

        // Self-reschedule loop.
        for {
            if err := sleepFn(ctx, Interval); err != nil {
                slog.Info("heartbeat goroutine exiting", "err", err)
                return
            }
            tick()
        }
    }

    // activeChars returns the union of char_names known from inventory and
    // spellbook mtime maps. Phase 1 had no LastKnownSpellbookMtime; Plan 02-02
    // added it. Either map being empty is fine.
    func activeChars(cfg *config.Config) []string {
        seen := map[string]bool{}
        for charName := range cfg.LastKnownInventoryMtime {
            seen[charName] = true
        }
        for charName := range cfg.LastKnownSpellbookMtime {
            seen[charName] = true
        }
        out := make([]string, 0, len(seen))
        for charName := range seen {
            out = append(out, charName)
        }
        return out
    }
    ```
