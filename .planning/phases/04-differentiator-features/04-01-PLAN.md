```typescript
// EverQuest Project 1999 constants. Single source of truth for class
// + race lists, slot vocabulary mapping, and any other static data
// shared across Phase 4+ triggers/builders.

export const CLASSES = [
  'WAR', 'CLR', 'PAL', 'RNG', 'SHD', 'DRU', 'MNK', 'BRD',
  'ROG', 'SHM', 'NEC', 'WIZ', 'MAG', 'ENC',
] as const;
export type ClassAbbrev = typeof CLASSES[number];

// Maps wiki class display name (e.g. "Necromancer") to 3-letter abbrev.
export const CLASS_DISPLAY_TO_ABBREV: Record<string, ClassAbbrev> = {
  'Bard': 'BRD', 'Cleric': 'CLR', 'Druid': 'DRU', 'Enchanter': 'ENC',
  'Magician': 'MAG', 'Monk': 'MNK', 'Necromancer': 'NEC', 'Paladin': 'PAL',
  'Ranger': 'RNG', 'Rogue': 'ROG', 'Shadow Knight': 'SHD',
  'Shaman': 'SHM', 'Warrior': 'WAR', 'Wizard': 'WIZ',
};

export const RACES = [
  'HUM', 'BAR', 'ERU', 'ELF', 'HIE', 'DEF', 'HEF', 'DWF',
  'TRL', 'OGR', 'HFL', 'GNM', 'IKS', 'VAH',
] as const;
export type RaceAbbrev = typeof RACES[number];

// Wiki gear-tier pages use prose slot labels; inv tab Slot column uses
// EQ in-game tokens. This map is many-to-many for pair slots — gear_check
// considers a char OK if EITHER slot in the array has the matching item.
export const WIKI_SLOT_TO_INV_SLOTS: Record<string, string[]> = {
  'Ears': ['EAR1', 'EAR2'],
  'Fingers': ['FINGER1', 'FINGER2'],
  'Wrists': ['WRIST1', 'WRIST2'],
  'Neck': ['NECK'], 'Head': ['HEAD'], 'Face': ['FACE'],
  'Chest': ['CHEST'], 'Arms': ['ARMS'], 'Back': ['BACK'],
  'Waist': ['WAIST'], 'Shoulders': ['SHOULDERS'], 'Legs': ['LEGS'],
  'Hands': ['HANDS'], 'Feet': ['FEET'],
  'Primary': ['PRIMARY'], 'Secondary': ['SECONDARY'], 'Range': ['RANGE'],
};

export function isClassAbbrev(s: unknown): s is ClassAbbrev {
  return typeof s === 'string' && (CLASSES as readonly string[]).includes(s);
}
export function isRaceAbbrev(s: unknown): s is RaceAbbrev {
  return typeof s === 'string' && (RACES as readonly string[]).includes(s);
}
```

```typescript
const CHAR_OWNER_V3_COLUMNS = ['race'];

export function migrateToV3(): MigrationOutcome {
  const current = readMetaRowInt('_meta', 'schema_version');
  if (current === 3) return 'noop_already_v2';  // type rename or just reuse the literal
  if (current !== 2) {
    log('warn', 'migrateToV3', { skipped: 'unsupported_version', current });
    return 'noop_unsupported_version';
  }
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(30000)) {
    throw new Error('migrateToV3: could not acquire document lock within 30s');
  }
  try {
    const charAdded = appendColumns('_char_owner', CHAR_OWNER_V3_COLUMNS);
    writeMetaRow('_meta', 'schema_version', '3');
    log('info', 'migrateToV3', { done: true, charAdded });
    return 'migrated_v1_to_v2';  // type extension or rename outcome enum
  } finally {
    lock.releaseLock();
  }
}
```

### Task 4 — showCharInfoSidebar + getCharsForForm + saveCharInfo

`apps-script/src/triggers/showCharInfoSidebar.ts`:

```typescript
import { log } from '../lib/log';
import { CLASSES, RACES, isClassAbbrev, isRaceAbbrev } from '../lib/eq-constants';
import { getActiveSpreadsheet } from '../lib/sheet-helpers';

const CHAR_OWNER_TAB = '_char_owner';
// _char_owner column indices (1-based): char_name=1, class=5, level=6, race=14
const COL_CHAR_NAME = 1, COL_CLASS = 5, COL_LEVEL = 6, COL_RACE = 14;

export interface CharInfoRow {
  char_name: string;
  class: string;
  level: number | string;  // string when blank
  race: string;
}

export function showCharInfoSidebar(): void {
  const html = HtmlService.createHtmlOutput(`<!-- inline HTML, see below -->`)
    .setTitle('Set Character Info')
    .setWidth(360);
  SpreadsheetApp.getUi().showSidebar(html);
}

export function getCharsForForm(): CharInfoRow[] {
  const sheet = getActiveSpreadsheet().getSheetByName(CHAR_OWNER_TAB);
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  const values = sheet.getRange(2, 1, lastRow - 1, COL_RACE).getValues();
  return values
    .map((r) => ({
      char_name: String(r[COL_CHAR_NAME - 1] ?? '').trim(),
      class: String(r[COL_CLASS - 1] ?? '').trim(),
      level: typeof r[COL_LEVEL - 1] === 'number' ? r[COL_LEVEL - 1] : String(r[COL_LEVEL - 1] ?? ''),
      race: String(r[COL_RACE - 1] ?? '').trim(),
    }))
    .filter((c) => c.char_name);
}

export function saveCharInfo(chars: CharInfoRow[]): { saved: number; errors: string[] } {
  const errors: string[] = [];
  for (const c of chars) {
    if (!c.char_name) { errors.push('Empty char_name'); continue; }
    if (c.class && !isClassAbbrev(c.class)) {
      errors.push(`${c.char_name}: invalid class ${c.class}`); continue;
    }
    const lvl = typeof c.level === 'number' ? c.level : parseInt(String(c.level || '0'), 10);
    if (c.level !== '' && (lvl < 1 || lvl > 60 || !Number.isFinite(lvl))) {
      errors.push(`${c.char_name}: level out of range`); continue;
    }
    if (c.race && !isRaceAbbrev(c.race)) {
      errors.push(`${c.char_name}: invalid race ${c.race}`); continue;
    }
  }
  if (errors.length) return { saved: 0, errors };

  const sheet = getActiveSpreadsheet().getSheetByName(CHAR_OWNER_TAB);
  if (!sheet) throw new Error('_char_owner missing — run migrateToV3 first');
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(30000)) throw new Error('saveCharInfo: lock_busy');
  try {
    // Read existing rows once
    const lastRow = sheet.getLastRow();
    const existing = lastRow >= 2
      ? sheet.getRange(2, 1, lastRow - 1, COL_RACE).getValues()
      : [];
    let saved = 0;
    for (const c of chars) {
      const idx = existing.findIndex((r) => String(r[0]).trim() === c.char_name);
      if (idx >= 0) {
        // Update class (col 5), level (col 6), race (col 14) in place
        if (c.class) sheet.getRange(idx + 2, COL_CLASS).setValue(c.class);
        if (c.level !== '') sheet.getRange(idx + 2, COL_LEVEL).setValue(c.level);
        if (c.race) sheet.getRange(idx + 2, COL_RACE).setValue(c.race);
        saved++;
      }
      // Don't insert new char_owner rows from the sidebar — watcher owns
      // row creation. Sidebar only updates existing chars.
    }
    log('info', 'saveCharInfo', { saved });
    return { saved, errors: [] };
  } finally {
    lock.releaseLock();
  }
}
```
